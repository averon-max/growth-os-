import type {
  CloudflareGraphQLResponse,
  CloudflareDailyGroup,
  DailyTrafficData,
  TrafficByCountry,
  TrafficByDevice,
  TrafficByBotCategory,
} from "./types";

const CLOUDFLARE_GRAPHQL_ENDPOINT = "https://api.cloudflare.com/client/v4/graphql";

interface FetchTrafficParams {
  zoneId: string;
  apiToken: string;
  since: string;
  until: string;
}

export interface SkippedDay {
  date: string;
  reason: string;
}

export interface FetchTrafficResult {
  days: DailyTrafficData[];
  skippedDays: SkippedDay[];
}

const QUERY = `
  query GetTraffic($zoneTag: String!, $day: Date!) {
    viewer {
      zones(filter: { zoneTag: $zoneTag }) {
        groups: httpRequestsAdaptiveGroups(
          limit: 10000
          filter: { date: $day }
          orderBy: [date_ASC]
        ) {
          count
          dimensions {
            date
            clientCountryName
            clientDeviceType
            verifiedBotCategory
          }
          sum {
            visits
            edgeResponseBytes
          }
        }
      }
    }
  }
`;

function enumerateDates(since: string, until: string): string[] {
  const dates: string[] = [];
  const current = new Date(since + "T00:00:00Z");
  const end = new Date(until + "T00:00:00Z");
  while (current <= end) {
    dates.push(current.toISOString().slice(0, 10));
    current.setUTCDate(current.getUTCDate() + 1);
  }
  return dates;
}

async function fetchOneDay(
  zoneId: string,
  apiToken: string,
  day: string
): Promise<CloudflareDailyGroup[]> {
  const response = await fetch(CLOUDFLARE_GRAPHQL_ENDPOINT, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${apiToken}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      query: QUERY,
      variables: { zoneTag: zoneId, day },
    }),
  });

  const json: CloudflareGraphQLResponse = await response.json();

  if (json.errors?.length) {
    throw new Error(json.errors.map((e) => e.message).join("; "));
  }

  return json.data?.viewer?.zones?.[0]?.groups ?? [];
}

export async function fetchCloudflareTraffic({
  zoneId,
  apiToken,
  since,
  until,
}: FetchTrafficParams): Promise<FetchTrafficResult> {
  const days = enumerateDates(since, until);
  const allGroups: CloudflareDailyGroup[] = [];
  const skippedDays: SkippedDay[] = [];

  for (const day of days) {
    try {
      const groups = await fetchOneDay(zoneId, apiToken, day);
      allGroups.push(...groups);
    } catch (err) {
      skippedDays.push({
        date: day,
        reason: err instanceof Error ? err.message : "Unknown Cloudflare error",
      });
    }
  }

  return { days: aggregateByDate(allGroups), skippedDays };
}

function aggregateByDate(groups: CloudflareDailyGroup[]): DailyTrafficData[] {
  const byDate = new Map<string, DailyTrafficData>();

  for (const g of groups) {
    const date = g.dimensions.date;
    if (!byDate.has(date)) {
      byDate.set(date, {
        date,
        totalRequests: 0,
        totalVisits: 0,
        bandwidthBytes: 0,
        byCountry: [],
        byDevice: [],
        byBotCategory: [],
      });
    }
    const entry = byDate.get(date)!;
    entry.totalRequests += g.count;
    entry.totalVisits += g.sum.visits;
    entry.bandwidthBytes += g.sum.edgeResponseBytes;

    upsertCountry(entry.byCountry, g.dimensions.clientCountryName, g.count, g.sum.visits);
    upsertDevice(entry.byDevice, g.dimensions.clientDeviceType, g.count, g.sum.visits);
    upsertBotCategory(entry.byBotCategory, g.dimensions.verifiedBotCategory, g.count);
  }

  return Array.from(byDate.values()).sort((a, b) => a.date.localeCompare(b.date));
}

function upsertCountry(list: TrafficByCountry[], country: string, requests: number, visits: number) {
  const existing = list.find((c) => c.country === country);
  if (existing) {
    existing.requests += requests;
    existing.visits += visits;
  } else {
    list.push({ country, requests, visits });
  }
}

function upsertDevice(list: TrafficByDevice[], device: string, requests: number, visits: number) {
  const existing = list.find((d) => d.device === device);
  if (existing) {
    existing.requests += requests;
    existing.visits += visits;
  } else {
    list.push({ device, requests, visits });
  }
}

function upsertBotCategory(list: TrafficByBotCategory[], category: string, requests: number) {
  const existing = list.find((b) => b.category === category);
  if (existing) {
    existing.requests += requests;
  } else {
    list.push({ category, requests });
  }
}

import { prisma } from "@/lib/prisma";
import { Prisma } from "@prisma/client";
import { fetchCloudflareTraffic, type SkippedDay } from "./client";

interface SyncParams {
  websiteId: string;
  zoneId: string;
  apiToken: string;
  since: string;
  until: string;
}

export interface SyncResult {
  connectionId: string;
  requestedDays: number;
  syncedDays: number;
  skippedDays: SkippedDay[];
  availableFrom: string | null;
  availableTo: string | null;
}

export async function syncCloudflareTraffic({
  websiteId,
  zoneId,
  apiToken,
  since,
  until,
}: SyncParams): Promise<SyncResult> {
  const connection = await prisma.cloudflareConnection.upsert({
    where: { websiteId },
    create: { websiteId, zoneId, apiToken },
    update: { zoneId, apiToken },
  });

  const { days: dailyTraffic, skippedDays } = await fetchCloudflareTraffic({
    zoneId,
    apiToken,
    since,
    until,
  });

  for (const day of dailyTraffic) {
    const totalUniqueVisitors = day.byCountry.reduce(
      (sum: number, c: { visits: number }) => sum + c.visits,
      0
    );

    const sortedCountries = [...day.byCountry].sort((a, b) => b.requests - a.requests);
    const sortedDevices = [...day.byDevice].sort((a, b) => b.requests - a.requests);
    const sortedBotCategories = [...day.byBotCategory].sort((a, b) => b.requests - a.requests);

    const verifiedBotRequests = day.byBotCategory
      .filter((b: { category: string }) => b.category !== "")
      .reduce((sum: number, b: { requests: number }) => sum + b.requests, 0);
    const notFlaggedAsBotRequests = day.totalRequests - verifiedBotRequests;

    await prisma.trafficSnapshot.upsert({
      where: {
        connectionId_date: {
          connectionId: connection.id,
          date: new Date(day.date),
        },
      },
      create: {
        connectionId: connection.id,
        date: new Date(day.date),
        totalRequests: day.totalRequests,
        humanRequests: notFlaggedAsBotRequests, // not identified as a known bot (best available on Free plan; may include unverified/stealth bots)
        verifiedBotRequests: verifiedBotRequests, // Cloudflare-identified crawlers (search engines, AI crawlers, SEO tools, etc.)
        unverifiedBotRequests: null, // Free plan does not expose this — unknown, not zero
        threatsBlocked: null, // Free plan does not expose this — unknown, not zero
        bandwidthBytes: BigInt(day.bandwidthBytes),
        uniqueVisitors: totalUniqueVisitors,
        countryBreakdown: sortedCountries as unknown as Prisma.InputJsonValue,
        deviceBreakdown: sortedDevices as unknown as Prisma.InputJsonValue,
        botCategoryBreakdown: sortedBotCategories as unknown as Prisma.InputJsonValue,
      },
      update: {
        totalRequests: day.totalRequests,
        humanRequests: notFlaggedAsBotRequests,
        verifiedBotRequests: verifiedBotRequests,
        unverifiedBotRequests: null,
        threatsBlocked: null,
        bandwidthBytes: BigInt(day.bandwidthBytes),
        uniqueVisitors: totalUniqueVisitors,
        countryBreakdown: sortedCountries as unknown as Prisma.InputJsonValue,
        deviceBreakdown: sortedDevices as unknown as Prisma.InputJsonValue,
        botCategoryBreakdown: sortedBotCategories as unknown as Prisma.InputJsonValue,
      },
    });
  }

  const syncedDates = dailyTraffic.map((d) => d.date).sort();

  return {
    connectionId: connection.id,
    requestedDays: Math.round(
      (new Date(until).getTime() - new Date(since).getTime()) / 86_400_000
    ) + 1,
    syncedDays: dailyTraffic.length,
    skippedDays,
    availableFrom: syncedDates[0] ?? null,
    availableTo: syncedDates[syncedDates.length - 1] ?? null,
  };
}

import { NextRequest, NextResponse } from "next/server";
import { syncCloudflareTraffic } from "@/lib/integrations/cloudflare/sync";

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const { websiteId, zoneId, apiToken, days = 7 } = body;

    if (!websiteId || !zoneId || !apiToken) {
      return NextResponse.json(
        { error: "websiteId, zoneId, and apiToken are required" },
        { status: 400 }
      );
    }

    const until = new Date().toISOString().slice(0, 10);
    const sinceDate = new Date();
    sinceDate.setDate(sinceDate.getDate() - days);
    const since = sinceDate.toISOString().slice(0, 10);

    const result = await syncCloudflareTraffic({ websiteId, zoneId, apiToken, since, until });

    return NextResponse.json({ success: true, ...result });
  } catch (error) {
    console.error("Cloudflare sync error:", error);
    return NextResponse.json(
      { error: error instanceof Error ? error.message : "Unknown error" },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function GET(req: NextRequest) {
  const websiteId = req.nextUrl.searchParams.get("websiteId");
  const days = Number(req.nextUrl.searchParams.get("days") ?? "7");

  if (!websiteId) {
    return NextResponse.json({ error: "websiteId is required" }, { status: 400 });
  }

  const connection = await prisma.cloudflareConnection.findUnique({
    where: { websiteId },
  });

  if (!connection) {
    return NextResponse.json({ connected: false, snapshots: [] });
  }

  const since = new Date();
  since.setDate(since.getDate() - days);

  const snapshots = await prisma.trafficSnapshot.findMany({
    where: { connectionId: connection.id, date: { gte: since } },
    orderBy: { date: "asc" },
  });

  const serialized = snapshots.map((s) => ({
    date: s.date.toISOString().slice(0, 10),
    totalRequests: s.totalRequests,
    humanRequests: s.humanRequests,
    verifiedBotRequests: s.verifiedBotRequests,
    uniqueVisitors: s.uniqueVisitors,
    bandwidthBytes: s.bandwidthBytes.toString(),
    countryBreakdown: s.countryBreakdown,
    deviceBreakdown: s.deviceBreakdown,
    botCategoryBreakdown: s.botCategoryBreakdown,
  }));

  return NextResponse.json({ connected: true, snapshots: serialized });
}
